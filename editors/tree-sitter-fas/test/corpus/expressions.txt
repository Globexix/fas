================================================================================
binary precedence

fn ops(a u32, b u32) bool {
  c u32 = a + b * 2 - a % b / 3
  d bool = a < b && !(a == b) || a >= 1
  e u32 = a | b ^ a & b
  f u32 = a << 2 >> 1
  return c != 0 && e <= f
}

  left f32 = if b { 0.2 } else if !b { 0.5 } else { 0.3 }

  return if a == 0 { left } else { right }
