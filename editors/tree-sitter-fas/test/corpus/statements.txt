fn loops(x i32) i32 {
  if x == 0 { return 1 }
  if x == 1 { return 2 } else { return 3 }
  if x == 2 { return 4 } else if x == 3 { return 5 } else { return 6 }
  while x > 0 { x -= 1 }
  for i usize = 0; i < 10; i += 1 { x += 1 }
  for ; ; { break }
  for i usize = 0; ; i = i + 1 { continue }
  return x
}
